cmake_minimum_required(VERSION 3.15)
project(sfxchat_helper C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(INSTALL_DOCS OFF CACHE BOOL "" FORCE)
set(INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(INSTALL_CMAKE_PACKAGE_MODULE OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${SOURCES}/libogg-1.3.6" ogg)
set(OGG_INCLUDE_DIR "${SOURCES}/libogg-1.3.6/include" CACHE PATH "" FORCE)
set(OGG_LIBRARY ogg CACHE STRING "" FORCE)
add_subdirectory("${SOURCES}/libvorbis-1.3.7" vorbis)
add_executable(sfxchat-helper helper.cpp)
target_compile_features(sfxchat-helper PRIVATE cxx_std_17)
target_link_libraries(sfxchat-helper PRIVATE vorbisenc vorbis ogg crypt32 advapi32 user32)
target_compile_definitions(sfxchat-helper PRIVATE _CRT_SECURE_NO_WARNINGS)
